cmake_minimum_required(VERSION 3.20)
project(HighPerformanceDataEngines VERSION 1.0.0 LANGUAGES CXX)
option(HPC_ENABLE_TESTS "Build integration tests" ON)
option(HPC_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
find_package(Threads REQUIRED)
function(add_engine name source)
  add_executable(${name} ${source})
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${name} PRIVATE Threads::Threads)
  if(HPC_ENABLE_SANITIZERS)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
add_engine(log_ingest src/log_ingest/main.cpp)
add_engine(text_search src/text_search/main.cpp)
add_engine(csv_aggregate src/csv_aggregate/main.cpp)
add_engine(shortest_path src/shortest_path/main.cpp)
if(HPC_ENABLE_TESTS)
  enable_testing()
  add_executable(runtime_unit tests/test_runtime.cpp)
  target_include_directories(runtime_unit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  add_test(NAME runtime_unit COMMAND runtime_unit)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py
      --bin-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(integration PROPERTIES TIMEOUT 120)
endif()
